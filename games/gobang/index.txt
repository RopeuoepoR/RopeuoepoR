    <script>
        class GomokuAI {
            constructor() {
                this.board = Array(15).fill().map(() => Array(15).fill(0));
                this.currentPlayer = 1; // 1=黑(玩家), 2=白(AI)
                this.gameOver = false;
                this.learningMode = false;
                this.learnedPatterns = new Map();
                this.gameHistory = [];
                this.stats = {
                    totalGames: 0,
                    playerWins: 0,
                    aiWins: 0,
                    trainCount: 0
                };
                this.loadStats();
            }

            initializeBoard() {
                const boardElement = document.getElementById('board');
                boardElement.innerHTML = '';
                
                for (let i = 0; i < 15; i++) {
                    for (let j = 0; j < 15; j++) {
                        const cell = document.createElement('div');
                        cell.className = 'cell';
                        cell.dataset.row = i;
                        cell.dataset.col = j;
                        cell.addEventListener('click', () => this.makeMove(i, j));
                        boardElement.appendChild(cell);
                    }
                }
            }

            makeMove(row, col) {
                if (this.gameOver || this.board[row][col] !== 0) return;
                
                this.board[row][col] = this.currentPlayer;
                this.gameHistory.push({row, col, player: this.currentPlayer});
                this.updateBoardDisplay();
                
                if (this.checkWin(row, col)) {
                    this.endGame(this.currentPlayer === 1 ? '🎉 玩家获胜！' : '🤖 AI获胜！');
                    return;
                }
                
                if (this.checkDraw()) {
                    this.endGame('🤝 平局！');
                    return;
                }
                
                this.currentPlayer = this.currentPlayer === 1 ? 2 : 1;
                this.updateStatus();
                
                if (this.currentPlayer === 2) {
                    setTimeout(() => this.aiMove(), 500);
                }
            }

            aiMove() {
                if (this.gameOver) return;
                
                let move;
                if (this.learningMode && this.learnedPatterns.size > 0) {
                    move = this.getSmartMove();
                } else {
                    move = this.getRandomMove();
                }
                
                if (move) {
                    this.makeMove(move.row, move.col);
                }
            }

            getSmartMove() {
                // 优先选择中心及附近位置
                const availableMoves = [];
                for (let i = 0; i < 15; i++) {
                    for (let j = 0; j < 15; j++) {
                        if (this.board[i][j] === 0) {
                            availableMoves.push({row: i, col: j});
                        }
                    }
                }
                
                availableMoves.sort((a, b) => {
                    const distA = Math.abs(a.row - 7) + Math.abs(a.col - 7);
                    const distB = Math.abs(b.row - 7) + Math.abs(b.col - 7);
                    return distA - distB;
                });
                
                return availableMoves[0];
            }

            getRandomMove() {
                const availableMoves = [];
                for (let i = 0; i < 15; i++) {
                    for (let j = 0; j < 15; j++) {
                        if (this.board[i][j] === 0) {
                            availableMoves.push({row: i, col: j});
                        }
                    }
                }
                
                if (availableMoves.length === 0) return null;
                return availableMoves[Math.floor(Math.random() * availableMoves.length)];
            }

            checkWin(row, col) {
                const directions = [
                    [[0, 1], [0, -1]], // 横向
                    [[1, 0], [-1, 0]], // 纵向
                    [[1, 1], [-1, -1]], // 对角线
                    [[1, -1], [-1, 1]] // 反对角线
                ];
                
                const player = this.board[row][col];
                
                for (const direction of directions) {
                    let count = 1;
                    
                    for (const [dr, dc] of direction) {
                        let r = row + dr;
                        let c = col + dc;
                        
                        while (r >= 0 && r < 15 && c >= 0 && c < 15 && this.board[r][c] === player) {
                            count++;
                            r += dr;
                            c += dc;
                        }
                    }
                    
                    if (count >= 5) return true;
                }
                
                return false;
            }

            checkDraw() {
                for (let i = 0; i < 15; i++) {
                    for (let j = 0; j < 15; j++) {
                        if (this.board[i][j] === 0) return false;
                    }
                }
                return true;
            }

            endGame(message) {
                this.gameOver = true;
                this.updateStatus(message);
                
                if (message.includes('AI')) {
                    this.stats.aiWins++;
                } else if (message.includes('玩家')) {
                    this.stats.playerWins++;
                }
                this.stats.totalGames++;
                
                if (this.learningMode) {
                    this.learnFromGame();
                }
                
                this.saveStats();
                this.updateStats();
            }

            learnFromGame() {
                // 记录最近3步作为学习模式
                if (this.gameHistory.length > 0) {
                    const pattern = this.gameHistory.slice(-3).map(move => 
                        `${move.row},${move.col},${move.player}`
                    ).join('|');
                    
                    this.learnedPatterns.set(pattern, (this.learnedPatterns.get(pattern) || 0) + 1);
                    document.getElementById('learnedPatterns').textContent = this.learnedPatterns.size;
                }
            }

            updateBoardDisplay() {
                for (let i = 0; i < 15; i++) {
                    for (let j = 0; j < 15; j++) {
                        const cell = document.querySelector(`[data-row="${i}"][data-col="${j}"]`);
                        cell.innerHTML = '';
                        
                        if (this.board[i][j] !== 0) {
                            const piece = document.createElement('div');
                            piece.className = `piece ${this.board[i][j] === 1 ? 'black' : 'white'}`;
                            cell.appendChild(piece);
                        }
                    }
                }
            }

            updateStatus(message) {
                const statusElement = document.getElementById('status');
                if (message) {
                    statusElement.textContent = message;
                } else if (this.gameOver) {
                    statusElement.textContent = '游戏结束';
                } else {
                    statusElement.textContent = `当前玩家: ${this.currentPlayer === 1 ? '黑子(玩家)' : '白子(AI)'}`;
                }
            }

            updateStats() {
                document.getElementById('totalGames').textContent = this.stats.totalGames;
                document.getElementById('playerWins').textContent = this.stats.playerWins;
                document.getElementById('aiWins').textContent = this.stats.aiWins;
                
                const winRate = this.stats.totalGames > 0 ? 
                    Math.round((this.stats.aiWins / this.stats.totalGames) * 100) : 0;
                document.getElementById('aiWinRate').textContent = winRate + '%';
                
                document.getElementById('trainCount').textContent = this.stats.trainCount;
                document.getElementById('learnedPatterns').textContent = this.learnedPatterns.size;
            }

            saveStats() {
                localStorage.setItem('gomokuStats', JSON.stringify(this.stats));
                localStorage.setItem('gomokuPatterns', JSON.stringify([...this.learnedPatterns]));
            }

            loadStats() {
                const savedStats = localStorage.getItem('gomokuStats');
                if (savedStats) {
                    this.stats = JSON.parse(savedStats);
                }
                
                const savedPatterns = localStorage.getItem('gomokuPatterns');
                if (savedPatterns) {
                    this.learnedPatterns = new Map(JSON.parse(savedPatterns));
                }
                
                this.updateStats();
            }

            toggleLearning() {
                this.learningMode = !this.learningMode;
                const btn = document.getElementById('learningBtn');
                const indicator = document.getElementById('learningIndicator');
                const status = document.getElementById('learningStatus');
                
                if (this.learningMode) {
                    btn.textContent = '关闭学习模式';
                    btn.className = 'btn btn-danger';
                    indicator.className = 'status-indicator active';
                    status.textContent = '学习模式已开启';
                } else {
                    btn.textContent = '启用学习模式';
                    btn.className = 'btn btn-success';
                    indicator.className = 'status-indicator';
                    status.textContent = '学习模式已关闭';
                }
            }

            trainAI() {
                this.stats.trainCount++;
                this.updateStatus('AI训练中...');
                
                setTimeout(() => {
                    // 模拟训练，添加随机模式
                    for (let i = 0; i < 10; i++) {
                        const pattern = Math.random().toString(36).substring(7);
                        this.learnedPatterns.set(pattern, Math.floor(Math.random() * 5) + 1);
                    }
                    
                    this.saveStats();
                    this.updateStats();
                    this.updateStatus('✅ AI训练完成！');
                    
                    setTimeout(() => this.updateStatus(), 2000);
                }, 2000);
            }

            resetAI() {
                if (confirm('确定要重置AI的所有学习数据吗？')) {
                    this.learnedPatterns.clear();
                    this.stats.trainCount = 0;
                    this.saveStats();
                    this.updateStats();
                    alert('AI知识已重置！');
                }
            }

            newGame() {
                this.board = Array(15).fill().map(() => Array(15).fill(0));
                this.currentPlayer = 1;
                this.gameOver = false;
                this.gameHistory = [];
                this.updateBoardDisplay();
                this.updateStatus();
            }
        }

        // 初始化游戏
        const game = new GomokuAI();
        game.initializeBoard();
        game.updateStatus();

        // 按钮点击全局函数
        function newGame() {
            game.newGame();
        }
        function toggleLearning() {
            game.toggleLearning();
        }
        function trainAI() {
            game.trainAI();
        }
        function resetAI() {
            game.resetAI();
        }
    </script>
